CREATE TABLE ST_product_departments (
    product_department_id INT AUTO_INCREMENT PRIMARY KEY,
    department_name_ar VARCHAR(200) NOT NULL,
    department_name_en VARCHAR(200) NOT NULL,
    image_url VARCHAR(500),
    icon VARCHAR(100),
    sort_order INT DEFAULT 0,
    is_active BIT DEFAULT 1,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE ST_product_categories (
    product_category_id INT AUTO_INCREMENT PRIMARY KEY,
    product_department_id INT NOT NULL,
    category_name_ar VARCHAR(200) NOT NULL,
    category_name_en VARCHAR(200) NOT NULL,
    image_url VARCHAR(500),
    sort_order INT DEFAULT 0,
    is_active BIT DEFAULT 1,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT FK_Category_Department
        FOREIGN KEY (product_department_id)
        REFERENCES ST_product_departments(product_department_id)
);

CREATE TABLE ST_product_subcategories (
    subcategory_id INT AUTO_INCREMENT PRIMARY KEY,
    product_category_id INT NOT NULL,
    subcategory_name_ar VARCHAR(200) NOT NULL,
    subcategory_name_en VARCHAR(200) NOT NULL,
    image_url VARCHAR(500),
    sort_order INT DEFAULT 0,
    is_active BIT DEFAULT 1,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT FK_SubCategory_Category
        FOREIGN KEY(product_category_id)
        REFERENCES ST_product_categories(product_category_id)
);

CREATE TABLE ST_product_brands (
    brand_id INT AUTO_INCREMENT PRIMARY KEY,
    brand_code VARCHAR(50) UNIQUE,
    brand_name_ar VARCHAR(200),
    brand_name_en VARCHAR(200),
    logo VARCHAR(500),
    website VARCHAR(300),
    country_id INT comment 'بلد المنشاء',
    is_active BIT DEFAULT 1 , 
	created_at DATETIME DEFAULT CURRENT_TIMESTAMP,   
        CONSTRAINT FK_country_id_Brand
        FOREIGN KEY(country_id)
        REFERENCES st_countries(country_id)
);

attribute_name_ar VARCHAR(100),
attribute_name_en VARCHAR(100),
attribute_icon  VARCHAR(100) ,
created_at DATETIME DEFAULT CURRENT_TIMESTAMP
